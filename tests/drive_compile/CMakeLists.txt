cmake_minimum_required(VERSION 3.24)
project(SocietyDriveCompile LANGUAGES CXX)

find_package(Qt6 6.8.3 EXACT REQUIRED COMPONENTS Core Gui Concurrent Qml)
get_filename_component(society_root "${CMAKE_CURRENT_SOURCE_DIR}/../.." ABSOLUTE)
get_filename_component(workspace "${society_root}/../.." ABSOLUTE)
set(SOCIETY_ACCOUNT_INCLUDE "" CACHE PATH "Installed account SDK headers")
if(NOT EXISTS "${SOCIETY_ACCOUNT_INCLUDE}/iiAcountManager/AccountManager.h")
    message(FATAL_ERROR "Set SOCIETY_ACCOUNT_INCLUDE to the account SDK include directory")
endif()
add_library(SocietyDriveCompile OBJECT "${society_root}/src/App/Drive/DriveController.cpp")
target_compile_features(SocietyDriveCompile PRIVATE cxx_std_23)
target_include_directories(SocietyDriveCompile PRIVATE "${workspace}/SDK/iiSocietyContainer" "${SOCIETY_ACCOUNT_INCLUDE}")
if(ANDROID)
    target_include_directories(SocietyDriveCompile PRIVATE "${workspace}/SDK/iiSocietyContainer/platform/android")
endif()
target_link_libraries(SocietyDriveCompile PRIVATE Qt6::Core Qt6::Gui Qt6::Concurrent Qt6::Qml)
if(WIN32)
    target_compile_definitions(SocietyDriveCompile PRIVATE SOCIETY_MOUNT_EXECUTABLE="iiSocietyContainerMount.exe")
endif()
