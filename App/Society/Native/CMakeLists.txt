cmake_minimum_required(VERSION 3.31)
project(SocietyNativeViews LANGUAGES CXX)
include(CTest)
find_package(iiSDKCore 0.20 CONFIG REQUIRED COMPONENTS Runtime)
find_package(iiJson 0.1 CONFIG REQUIRED)
find_package(iiNetwork 0.6 CONFIG REQUIRED)
find_package(iiGUI 0.5 CONFIG REQUIRED)
find_package(LVUI 0.1 CONFIG REQUIRED COMPONENTS Native)
add_library(SocietyNativeBackend NativeBackendWire.cpp NativeBackendConnection.cpp)
target_compile_features(SocietyNativeBackend PUBLIC cxx_std_23)
target_include_directories(SocietyNativeBackend PUBLIC "${CMAKE_CURRENT_SOURCE_DIR}/../..")
target_link_libraries(SocietyNativeBackend PUBLIC iiJson::iiJson iiNetwork::iiNetwork)
add_library(SocietyNativeFilesViewModel NativeFilesViewModel.cpp)
target_link_libraries(SocietyNativeFilesViewModel PUBLIC SocietyNativeBackend LVUI::LVUI LVUI::Native iiGUI::iiGUI)
lvui_generate(NAME NativeFiles SOURCE NativeFiles.lvui INCLUDES NativeFilesViewModel.h NativeFilesNavigation.h OUTPUT_VARIABLE native_files_view)
add_custom_target(SocietyNativeFilesGenerated DEPENDS "${native_files_view}")
iisdkcore_add_application(SocietyNativeFilesMigration
    SOURCES NativeFilesApplication.cpp
    LIBRARIES SocietyNativeFilesViewModel iiGUI::iiGUI LVUI::LVUI LVUI::Native)
if(APPLE AND NOT IOS)
    enable_language(OBJCXX)
    target_sources(SocietyNativeFilesMigration PRIVATE NativeFileOpenMac.mm NativeFileOpen.h)
    set_target_properties(SocietyNativeFilesMigration PROPERTIES OBJCXX_STANDARD 23 OBJCXX_STANDARD_REQUIRED YES)
    target_link_libraries(SocietyNativeFilesMigration PRIVATE "-framework AppKit")
else()
    target_sources(SocietyNativeFilesMigration PRIVATE NativeFileOpenUnsupported.cpp NativeFileOpen.h)
endif()
add_dependencies(SocietyNativeFilesMigration SocietyNativeFilesGenerated)
target_include_directories(SocietyNativeFilesMigration PRIVATE "${CMAKE_CURRENT_SOURCE_DIR}" "${CMAKE_CURRENT_BINARY_DIR}")
target_compile_features(SocietyNativeFilesMigration PRIVATE cxx_std_23)
if(BUILD_TESTING)
    add_executable(SocietyNativeBackendWireTests ../../../tests/native_backend_wire.cpp)
    target_link_libraries(SocietyNativeBackendWireTests PRIVATE SocietyNativeBackend)
    add_test(NAME Society.NativeBackendWire COMMAND SocietyNativeBackendWireTests)
endif()
if(BUILD_TESTING AND UNIX)
    find_package(Qt6 6.8.3 EXACT CONFIG REQUIRED COMPONENTS Core Network)
    find_package(iiSocietyContainer 0.16.1 CONFIG REQUIRED)
    add_executable(SocietyNativeBackendConnectionTests ../../../tests/native_backend_connection.cpp)
    target_compile_definitions(SocietyNativeBackendConnectionTests PRIVATE
        SOCIETY_NATIVE_RESOURCE_ROOT="${CMAKE_CURRENT_SOURCE_DIR}/../../../resources")
    target_link_libraries(SocietyNativeBackendConnectionTests PRIVATE SocietyNativeBackend)
    add_test(NAME Society.NativeBackendConnection COMMAND SocietyNativeBackendConnectionTests)
    add_executable(SocietyNativeFilesClientTests ../../../tests/native_files_client.cpp)
    target_compile_definitions(SocietyNativeFilesClientTests PRIVATE SOCIETY_NATIVE_RESOURCE_ROOT="${CMAKE_CURRENT_SOURCE_DIR}/../../../resources")
    target_link_libraries(SocietyNativeFilesClientTests PRIVATE SocietyNativeFilesViewModel)
    add_test(NAME Society.NativeFilesClient COMMAND SocietyNativeFilesClientTests)
    add_executable(SocietyNativeFilesIntegrationTests ../../../tests/native_files_integration.cpp
        ../../Daemon/ModelBackendServer.cpp ../../Daemon/ModelBackendServer.h
        ../../Daemon/NativeDirectorySession.cpp ../../Daemon/NativeDirectorySession.h)
    set_target_properties(SocietyNativeFilesIntegrationTests PROPERTIES AUTOMOC ON)
    target_compile_definitions(SocietyNativeFilesIntegrationTests PRIVATE SOCIETY_NATIVE_RESOURCE_ROOT="${CMAKE_CURRENT_SOURCE_DIR}/../../../resources")
    target_link_libraries(SocietyNativeFilesIntegrationTests PRIVATE SocietyNativeFilesViewModel
        iiSocietyContainer::iiSocietyContainer Qt6::Core Qt6::Network)
    add_test(NAME Society.NativeFilesIntegration COMMAND SocietyNativeFilesIntegrationTests)
    set_tests_properties(Society.NativeFilesIntegration PROPERTIES TIMEOUT 180)
endif()
if(BUILD_TESTING)
    add_executable(SocietyNativeFilesViewModelTests ../../../tests/native_files_viewmodel.cpp)
    target_link_libraries(SocietyNativeFilesViewModelTests PRIVATE SocietyNativeFilesViewModel)
    add_test(NAME Society.NativeFilesViewModel COMMAND SocietyNativeFilesViewModelTests)
    add_executable(SocietyNativeFilesViewCompile ../../../tests/native_files_view_compile.cpp)
    add_dependencies(SocietyNativeFilesViewCompile SocietyNativeFilesGenerated)
    target_include_directories(SocietyNativeFilesViewCompile PRIVATE "${CMAKE_CURRENT_SOURCE_DIR}" "${CMAKE_CURRENT_BINARY_DIR}")
    target_link_libraries(SocietyNativeFilesViewCompile PRIVATE SocietyNativeFilesViewModel)
    add_test(NAME Society.NativeFilesViewCompile COMMAND SocietyNativeFilesViewCompile)
endif()
